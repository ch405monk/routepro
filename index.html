function deployBatch(batchIdx) { if (activeRoute.length === 0) return alert('No active stops to navigate.'); const batchSize = 10; let startIdx = batchIdx * batchSize; let endIdx = Math.min((batchIdx + 1) * batchSize, activeRoute.length); let batchStops = activeRoute.slice(startIdx, endIdx); let startOrigin = (batchIdx === 0) ? encodeURIComponent(document.getElementById('startLocation').value || 'Copperfield, Calgary SE') : encodeURIComponent(activeRoute[startIdx - 1].address + ', Calgary, AB'); if (mapProvider === 'apple') { // Apple Maps requires individual &daddr= parameters for each waypoint in sequence let daddrs = batchStops.map(s => encodeURIComponent(s.address + ', Calgary, AB')).join('&daddr='); let url = `https://maps.apple.com/?saddr=${startOrigin}&daddr=${daddrs}`; window.open(url, '_blank'); } else { // Google Maps takes intermediate stops via &waypoints= separated by pipes (|) let destination = encodeURIComponent(batchStops[batchStops.length - 1].address + ', Calgary, AB'); let middleStops = batchStops.slice(0, -1); let url = `https://www.google.com/maps/dir/?api=1&origin=${startOrigin}&destination=${destination}`; if (middleStops.length > 0) { let waypoints = middleStops.map(s => encodeURIComponent(s.address + ', Calgary, AB')).join('|'); url += `&waypoints=${waypoints}`; } window.open(url, '_blank'); } }
